const countEl = document.getElementById('count'); const pauseBtn = document.getElementById('pause'); const likesKey = 'likes-me-total'; let likes = Number(localStorage.getItem(likesKey) || 0); let paused = false; let lastTick = Date.now(); function paint() { countEl.textContent = likes.toLocaleString('es-UY'); localStorage.setItem(likesKey, String(likes)); } // Mostrar el total guardado al entrar paint(); // Sumar 1 like por segundo setInterval(() => { if (!paused) { const now = Date.now(); const seconds = Math.floor((now - lastTick) / 1000); if (seconds >= 1) { likes += seconds; lastTick += seconds * 1000; paint(); } } else { lastTick = Date.now(); } }, 200); // Pausar y continuar pauseBtn.addEventListener('click', () => { paused = !paused; pauseBtn.textContent = paused ? 'Continuar' : 'Pausar'; lastTick = Date.now(); });
